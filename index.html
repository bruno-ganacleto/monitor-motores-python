Monitoramento de Motores em Python

Documentação técnica, parâmetros operacionais monitorados e implementação de telemetria e análise de falhas em tempo real.

Acessar Repositório no GitHub

1. Explicação Técnica

O sistema foi projetado para realizar a aquisição, o processamento e o diagnóstico contínuo das condições operacionais de motores elétricos industriais. Utilizando Python como núcleo de processamento, a aplicação realiza a leitura cíclica de sensores (ou simulações de telemetria via protocolos industriais como Modbus/MQTT) e compara as grandezas medidas contra limites nominais de segurança.

Aquisição de Dados

Leitura periódica dos sensores acoplados à carcaça, mancais e quadro de alimentação do motor.

Análise de Limiares

Classificação automática do estado operacional em Normal, Alerta ou Crítico com base em faixas de tolerância.

Registro e Alarme

Geração de logs estruturados com carimbo de data/hora (timestamp) e disparo preventivo de alertas para a equipe de manutenção.

import time
import random
from datetime import datetime

# Limites operacionais configurados para o motor
LIMITES = {
    "temperatura": {"alerta": 75.0, "critico": 85.0, "unidade": "°C"},
    "vibracao":    {"alerta": 2.8,  "critico": 4.5,  "unidade": "mm/s"},
    "corrente":    {"alerta": 18.0, "critico": 20.0, "unidade": "A"},
    "rpm":         {"min_alerta": 1700, "min_critico": 1650, "unidade": "rpm"}
}

def ler_sensores():
    """Simula a coleta de telemetria dos sensores do motor."""
    return {
        "temperatura": round(random.uniform(55.0, 90.0), 2),
        "vibracao":    round(random.uniform(1.0, 5.0), 2),
        "corrente":    round(random.uniform(12.0, 21.5), 2),
        "rpm":         random.randint(1630, 1775)
    }

def avaliar_status(leituras):
    """Classifica a condição operacional com base nos limiares definidos."""
    alertas = []
    status_geral = "NORMAL"

    for param in ["temperatura", "vibracao", "corrente"]:
        valor = leituras[param]
        if valor >= LIMITES[param]["critico"]:
            status_geral = "CRÍTICO"
            alertas.append(f"{param.upper()} CRÍTICA ({valor} {LIMITES[param]['unidade']})")
        elif valor >= LIMITES[param]["alerta"]:
            if status_geral != "CRÍTICO":
                status_geral = "ALERTA"
            alertas.append(f"{param.upper()} EM ALERTA ({valor} {LIMITES[param]['unidade']})")

    if leituras["rpm"] <= LIMITES["rpm"]["min_critico"]:
        status_geral = "CRÍTICO"
        alertas.append(f"ROTAÇÃO BAIXA CRÍTICA ({leituras['rpm']} rpm)")
    elif leituras["rpm"] <= LIMITES["rpm"]["min_alerta"]:
        if status_geral != "CRÍTICO":
            status_geral = "ALERTA"
        alertas.append(f"ROTAÇÃO EM ALERTA ({leituras['rpm']} rpm)")

    return status_geral, alertas

def executar_monitoramento(ciclos=5, intervalo=2):
    print("Iniciando monitoramento de motores industriais...\n")
    for _ in range(ciclos):
        timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        dados = ler_sensores()
        status, ocorrencias = avaliar_status(dados)

        print(f"[{timestamp}] Status: {status} | Dados: {dados}")
        for item in ocorrencias:
            print(f"   -> Alarme: {item}")
        time.sleep(intervalo)

if __name__ == "__main__":
    executar_monitoramento()
